    <script>
        // --- Splash Particle Logic (Node Connectivity) ---
        (function () {
            const sc = document.getElementById('splash-canvas');
            const sctx = sc.getContext('2d');
            let sw, sh;
            let particles = [];
            let sMouse = { x: null, y: null };

            function resize() {
                sw = sc.width = window.innerWidth;
                sh = sc.height = window.innerHeight;
            }
            window.addEventListener('resize', resize);
            resize();

            // Text Spotlight Logic
            const sLogo = document.getElementById('s-logo');
            window.addEventListener('mousemove', (e) => {
                sMouse.x = e.clientX;
                sMouse.y = e.clientY;

                if (sLogo) {
                    const rect = sLogo.getBoundingClientRect();
                    const lx = e.clientX - rect.left;
                    const ly = e.clientY - rect.top;
                    sLogo.style.setProperty('--x', `${lx}px`);
                    sLogo.style.setProperty('--y', `${ly}px`);
                }
            });

            class SParticle {
                constructor() {
                    this.x = Math.random() * sw;
                    this.y = Math.random() * sh;
                    this.vx = (Math.random() - 0.5) * 0.4;
                    this.vy = (Math.random() - 0.5) * 0.4;
                    // ?üÁ©∫?®Ê?Â§ßÂ??áÂëº?∏È†ª??
                    this.radius = Math.random() * 1.5 + 0.5;
                    this.phase = Math.random() * Math.PI * 2;
                    this.phaseSpeed = 0.01 + Math.random() * 0.03;
                    this.baseOpacity = 0.1 + Math.random() * 0.4;
                }
                update() {
                    this.x += this.vx;
                    this.y += this.vy;
                    this.phase += this.phaseSpeed;

                    if (this.x < 0 || this.x > sw) this.vx *= -1;
                    if (this.y < 0 || this.y > sh) this.vy *= -1;

                    // Mouse interaction
                    if (sMouse.x) {
                        const dx = sMouse.x - this.x;
                        const dy = sMouse.y - this.y;
                        const dist = Math.sqrt(dx * dx + dy * dy);
                        if (dist < 150) {
                            this.x -= dx * 0.005;
                            this.y -= dy * 0.005;
                        }
                    }
                }
                draw() {
                    const breathing = (Math.sin(this.phase) + 1) / 2; // 0 ~ 1
                    const opacity = this.baseOpacity + (breathing * 0.3);

                    sctx.fillStyle = `rgba(200, 169, 126, ${opacity})`;
                    // ?πÂ§ß‰∫ÆÂ∫¶?ÉÊ??ÉÁ??âÊ???
                    if (this.baseOpacity > 0.4) {
                        sctx.shadowBlur = breathing * 8;
                        sctx.shadowColor = 'rgba(200, 169, 126, 0.5)';
                    } else {
                        sctx.shadowBlur = 0;
                    }

                    sctx.beginPath();
                    sctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                    sctx.fill();
                    sctx.shadowBlur = 0; // ?çÁΩÆ
                }
            }

            function init() {
                particles = [];
                // ?∏È?Â¢ûÂ??∫Â??àÁ? 1.5 ~ 2 ??
                const count = Math.min(150, (sw * sh) / 8000);
                for (let i = 0; i < count; i++) {
                    particles.push(new SParticle());
                }
            }
            init();

            function drawLines() {
                for (let i = 0; i < particles.length; i++) {
                    for (let j = i + 1; j < particles.length; j++) {
                        const dx = particles[i].x - particles[j].x;
                        const dy = particles[i].y - particles[j].y;
                        const dist = Math.sqrt(dx * dx + dy * dy);

                        if (dist < 160) {
                            const opacity = 1 - (dist / 160);
                            sctx.strokeStyle = `rgba(200, 169, 126, ${opacity * 0.15})`;
                            sctx.lineWidth = 0.8;
                            sctx.beginPath();
                            sctx.moveTo(particles[i].x, particles[i].y);
                            sctx.lineTo(particles[j].x, particles[j].y);
                            sctx.stroke();
                        }
                    }
                }
            }

            function loop() {
                if (document.getElementById('splash-screen').classList.contains('hidden')) return;
                sctx.clearRect(0, 0, sw, sh);
                particles.forEach(p => {
                    p.update();
                    p.draw();
                });
                drawLines();
                requestAnimationFrame(loop);
            }
            loop();
        })();

        // Splash Reveal Sequence
        window.addEventListener('load', () => {
            const ids = ['s-meta', 's-logo', 's-desc', 's-btn', 's-sub'];
            ids.forEach(id => {
                const el = document.getElementById(id);
                if (el) el.classList.add('reveal-node');
            });
        });

        (function() {
            const sBtn = document.getElementById('s-btn');
            sBtn.addEventListener('click', () => {
                document.getElementById('splash-screen').classList.add('hidden');
                // ?≠Êîæ?åÊôØ?≥Ê?
                const bgm = document.getElementById('bgm');
                if (bgm) bgm.play().catch(e => {});
            });
        })();
    </script>
